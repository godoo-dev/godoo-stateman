```python
from dataclasses import dataclass
SCHEMA_FORMAT_VERSION: int = 1

@dataclass(frozen=True)
class OdooVersion:
    major: int
    minor: int
    def __str__(self) -> str: return f"{self.major}.{self.minor}"

# src/godoo_stateman/types/schema.py
from pydantic import BaseModel, ConfigDict

class VersionedFieldSchema(BaseModel):
    model_config = ConfigDict(frozen=True)
    name: str
    ttype: str
    store: bool
    readonly: bool
    compute: str | None
    relation: str | None
    required: bool

class VersionedModelSchema(BaseModel):
    model_config = ConfigDict(frozen=True)
    name: str
    display_name: str
    transient: bool
    odoo_version: str  # str form of OdooVersion e.g. "17.0"
    archivable: bool   # True if model has a stored writable 'active' field
    fields: dict[str, VersionedFieldSchema]

# src/godoo_stateman/schema/registry.py
class SchemaRegistry:
    def __init__(self, client: OdooClient, version: OdooVersion) -> None: ...
    async def get(self, model_name: str, *, bypass_cache: bool = False) -> VersionedModelSchema: ...
    def cache_path(self, url: str, database: str) -> Path: ...
    async def build_snapshot(self, model_names: list[str]) -> VersionedSnapshot: ...

# src/godoo_stateman/schema/snapshot.py
class VersionedSnapshot(BaseModel):
    model_config = ConfigDict(frozen=True)
    odoo_version: str
    schema_format_version: int
    captured_at: str  # ISO 8601
    models: dict[str, VersionedModelSchema]
    def save(self, path: Path) -> None: ...
    @classmethod
    def load(cls, path: Path, expected_odoo_version: str) -> VersionedSnapshot: ...
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| env vars → OdooClientConfig | Credentials flow from OS environment into the client; never from the `.py` config file or CLI args |
| `ODOO_VERSION` env → snapshot file path | The Odoo version string is used as a path component in the cache directory; must be sanitized |
| snapshot JSON file → VersionedSnapshot | Pydantic model_validate() is the deserialization boundary; unknown fields must not panic |
| http URL → Odoo 17 | Unencrypted transport risks credential interception on non-local hosts |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-01 | Information Disclosure | cli/commands/snapshot.py `_snapshot_impl` | mitigate | Credentials read from env only (`ODOO_URL`, `ODOO_DB`, `ODOO_USER`, `ODOO_PASSWORD`); missing-var error prints var NAMES not values; never log credential values |
| T-02-02 | Tampering | schema/snapshot.py `VersionedSnapshot.load()` | mitigate | `Pydantic.model_validate()` rejects unknown fields and type violations with `ValidationError`; `VersionMismatchError` is raised before any schema data is trusted |
| T-02-03 | Tampering | `ODOO_VERSION` env → cache path segment | mitigate | Validate with `re.match(r"^\d+\.\d+$", odoo_version)` before use as path component; reject non-matching values with `typer.BadParameter` |
| T-02-04 | Information Disclosure | http:// URL to non-local Odoo | accept (warn) | ASVS L1: warn via Rich console when URL scheme is `http` and host is not `localhost`/`127.0.0.1`; blocking is out of scope at L1 |
| T-02-SC | Tampering | uv add / pip install | mitigate | slopcheck gate for `platformdirs` ([ASSUMED]); see T-01-SC in plan 01 |
